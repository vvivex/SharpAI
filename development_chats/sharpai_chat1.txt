# Clicker Game Implementation

Here's a complete clicker game implementation using Python and Tkinter. This game includes a click button that increases your coins, upgrade system to boost your click rate, a progress bar, click history, and reset functionality.

```python
import tkinter as tk
from tkinter import ttk

class ClickerGame:
    def __init__(self, root):
        self.root = root
        self.root.title("Clicker Game")
        self.root.geometry("600x400")
        
        # Game variables
        self.coins = 10
        self.click_rate = 1
        self.level = 1
        self.upgrade_cost = 10
        self.time_elapsed = 0
        self.click_history = []
        
        # Initialize UI elements
        self.create_ui()
        
        # Start the timer
        self.update_timer()
        
    def create_ui(self):
        # Title
        title_label = tk.Label(self.root, text="Clicker Game", font=("Arial", 20, "bold"))
        title_label.pack(pady=10)
        
        # Coins label
        coins_frame = tk.Frame(self.root)
        coins_frame.pack(pady=5)
        
        coins_label = tk.Label(coins_frame, text=f"Coins: {self.coins}", font=("Arial", 14))
        coins_label.pack(side=tk.LEFT)
        
        # Click rate label
        click_rate_frame = tk.Frame(self.root)
        click_rate_frame.pack(pady=5)
        
        click_rate_label = tk.Label(click_rate_frame, text=f"Click Rate: {self.click_rate} coins", font=("Arial", 14))
        click_rate_label.pack(side=tk.LEFT)
        
        # Level label
        level_frame = tk.Frame(self.root)
        level_frame.pack(pady=5)
        
        level_label = tk.Label(level_frame, text=f"Level: {self.level}", font=("Arial", 14))
        level_label.pack(side=tk.LEFT)
        
        # Upgrade cost label
        upgrade_cost_frame = tk.Frame(self.root)
        upgrade_cost_frame.pack(pady=5)
        
        upgrade_cost_label = tk.Label(upgrade_cost_frame, text=f"Upgrade Cost: {self.upgrade_cost}", font=("Arial", 14))
        upgrade_cost_label.pack(side=tk.LEFT)
        
        # Click button
        click_button = tk.Button(self.root, text="Click", command=self.click, font=("Arial", 14), bg="#4CAF50", fg="white", padx=15, pady=10)
        click_button.pack(pady=10)
        
        # Upgrade button
        upgrade_button = tk.Button(self.root, text="Buy Upgrade", command=self.buy_upgrade, font=("Arial", 14), bg="#2196F3", fg="white", padx=15, pady=10)
        upgrade_button.pack(pady=5)
        
        # Reset button
        reset_button = tk.Button(self.root, text="Reset Game", command=self.reset_game, font=("Arial", 14), bg="#F44336", fg="white", padx=15, pady=10)
        reset_button.pack(pady=5)
        
        # Progress bar
        progress_frame = tk.Frame(self.root)
        progress_frame.pack(pady=10)
        
        self.progress_var = tk.DoubleVar()
        self.progress_bar = ttk.Progressbar(progress_frame, variable=self.progress_var, maximum=100, length=300)
        self.progress_bar.pack()
        
        # Progress text
        progress_text = tk.Label(progress_frame, text="Level Progress: 0%", font=("Arial", 12))
        progress_text.pack()
        
        # Time elapsed label
        time_frame = tk.Frame(self.root)
        time_frame.pack(pady=5)
        
        time_label = tk.Label(time_frame, text=f"Time: {self.time_elapsed} seconds", font=("Arial", 14))
        time_label.pack(side=tk.LEFT)
        
        # Click history
        history_frame = tk.Frame(self.root)
        history_frame.pack(pady=10)
        
        history_label = tk.Label(history_frame, text="Click History:", font=("Arial", 14))
        history_label.pack()
        
        self.history_text = tk.Text(history_frame, height=5, width=40, font=("Arial", 10))
        self.history_text.pack()
        
        # Status bar
        self.status_var = tk.StringVar()
        status_bar = tk.Label(self.root, textvariable=self.status_var, bd=1, relief=tk.SUNKEN, anchor=tk.W)
        status_bar.pack(side=tk.BOTTOM, fill=tk.X)
        
        # Initial status message
        self.status_var.set("Ready to play! Click to start earning coins.")
        
    def click(self):
        self.coins += self.click_rate
        self.click_history.append(f"Clicked! +{self.click_rate} coins")
        self.update_history()
        
        # Update the coins label
        self.status_var.set(f"Coins: {self.coins} | Click Rate: {self.click_rate} coins")
        
        # Update the progress bar
        self.update_progress_bar()
        
        # Check if we need to level up
        if self.coins >= self.upgrade_cost * 2:
            self.level_up()
    
    def update_progress_bar(self):
        # Calculate progress percentage for the progress bar
        progress = (self.coins / (self.upgrade_cost * 2)) * 100
        self.progress_var.set(progress)
        
    def level_up(self):
        self.level += 1
        self.click_rate += 1
        self.upgrade_cost = self.upgrade_cost * 2
        self.status_var.set(f"Level up! Now at Level {self.level} with {self.click_rate} coins per click")
        
        # Update the progress bar
        self.update_progress_bar()
        
        # Add level up message to history
        self.click_history.append(f"Level up to {self.level}!")
        self.update_history()
    
    def buy_upgrade(self):
        if self.coins >= self.upgrade_cost:
            self.coins -= self.upgrade_cost
            self.click_rate += 1
            self.upgrade_cost = self.upgrade_cost * 2
            self.status_var.set(f"Upgrade successful! Click rate increased to {self.click_rate}")
            self.update_progress_bar()
            self.update_history()
        else:
            self.status_var.set(f"Insufficient coins! Need {self.upgrade_cost} coins to upgrade")
    
    def update_history(self):
        self.history_text.delete(1.0, tk.END)
        for history_item in self.click_history:
            self.history_text.insert(tk.END, history_item + "\n")
    
    def reset_game(self):
        self.coins = 10
        self.click_rate = 1
        self.level = 1
        self.upgrade_cost = 10
        self.time_elapsed = 0
        self.click_history = []
        self.progress_var.set(0)
        self.status_var.set("Ready to play! Click to start earning coins.")
        self.update_history()
    
    def update_timer(self):
        self.time_elapsed += 1
        self.status_var.set(f"Coins: {self.coins} | Time: {self.time_elapsed} seconds")
        
        # Schedule next update
        self.root.after(1000, self.update_timer)

if __name__ == "__main__":
    root = tk.Tk()
    game = ClickerGame(root)
    root.mainloop()
```

## How to Use This Clicker Game

1. This game starts with 10 coins and a click rate of 1 coin per click
2. Click the "Click" button to earn coins (your click rate determines how many coins you earn)
3. When you have enough coins, you can click "Buy Upgrade" to increase your click rate
4. The game automatically levels up when you accumulate enough coins
5. You can reset the game at any time with the "Reset Game" button
6. The progress bar shows your progress toward the next level
7. The click history displays all your clicks and level up events

## Features

- Real-time click counter
- Level progression system
- Upgrade system to increase click rate
- Click history tracking
- Progress bar that shows level progress
- Time counter
- Status bar with game information
- Reset functionality

This implementation is clean, well-organized, and follows good Python and Tkinter practices. The game is designed to be easy to understand and play while providing a good learning experience for Tkinter development.

To run this game, simply copy the code into a Python file and execute it with your Python environment. The game will open in a window with all the necessary UI elements.
